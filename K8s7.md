定义日志写文件接口file.go

```go
package logs
import (
	"ClusterManagement/config"
	"github.com/google/uuid"
	"os"
)
type fileWriter struct {
	*os.File
}
func (s *fileWriter) Flush() {
	s.Sync()
}
func NewFileWriter(config *config.Config) LogWriter {
	filename := uuid.New().String() + ".log"
	filePath := config.Log.Path + "/" + filename
	file, err := os.OpenFile(filePath, os.O_CREATE|os.O_WRONLY|os.O_APPEND, 0644)
	if err != nil {
		file = os.Stderr
	}
	return &fileWriter{
		file,
	}
}
```

定义日志写cls云服务中心cls.go：

```go
package logs
import (
	"ClusterManagement/config"
	"encoding/json"
	"fmt"
	cls "github.com/tencentcloud/tencentcloud-cls-sdk-go"
	"log"
	"time"
)
type clsWriter struct {
	config    *config.Config
	clsClient *cls.AsyncProducerClient
}
func NewClsWriter(config *config.Config) LogWriter {
	client, err := getClient(config)
	if err != nil {
		log.Fatal(err)
	}
	return &clsWriter{
		config:    config,
		clsClient: client,
	}
}
func (c *clsWriter) Write(p []byte) (n int, err error) {
	mp := make(map[string]string)
	err = json.Unmarshal(p, &mp)
	if err != nil {
		mp["content"] = string(p)
	}
	c.sendLog(mp)
	return 0, nil
}
func (c *clsWriter) Flush() {
	c.clsClient.Close(60000)
}
func (c *clsWriter) sendLog(msg map[string]string) {
	logMsg := cls.NewCLSLog(time.Now().Unix(), msg)
	err := c.clsClient.SendLog(c.config.Cls.TopicId, logMsg, &CallBack{})
	if err != nil {
		log.Println(err)
		return
	}
}
func getClient(config *config.Config) (*cls.AsyncProducerClient, error) {
	producerConfig := cls.GetDefaultAsyncProducerClientConfig()
	producerConfig.Endpoint = config.Cls.Endpoint
	producerConfig.AccessKeyID = config.Cls.SecretID
	producerConfig.AccessKeySecret = config.Cls.SecretKey
	producerInstance, err := cls.NewAsyncProducerClient(producerConfig)
	if err != nil {
		log.Println(err)
		return nil, err
	}
	producerInstance.Start()
	return producerInstance, nil
}
type CallBack struct{}
func (callback *CallBack) Success(result *cls.Result) {
	fmt.Println("cls 日志发送成功")
}
func (CallBack *CallBack) Fail(result *cls.Result) {
	fmt.Printf("cls 日志发送失败：%v \n", result.GetErrorMessage())
}
```

```go
	conf = &Config{}
	err := v.Unmarshal(conf)
	if err != nil {
		log.Fatal(err)
	}

func GetConf() *Config {
	return conf
```

```go
package app
import (
	"ClusterManagement/config"
	"github.com/gin-gonic/gin"
	"github.com/sirupsen/logrus"
)
type AppInterface interface {
	WriteLogsToFile(c *gin.Context)
	WriteLogsToCls(c *gin.Context)
	Ping(c *gin.Context)
	Health(c *gin.Context)
}

type app struct {
	config     *config.Config
	fileLogger *logrus.Entry
	clsLogger  *logrus.Entry
}

func NewApp(config *config.Config, fileLogger *logrus.Entry, clsLogger *logrus.Entry) AppInterface {
	return &app{
		config:     config,
		fileLogger: fileLogger,
		clsLogger:  clsLogger,
	}
}
```

实现接口的业务逻辑controller.go:

```go
package app
import (
	"github.com/gin-gonic/gin"
	"github.com/sirupsen/logrus"
	"net"
	"net/http"
	"strconv"
	"time"
)
func (a *app) WriteLogsToFile(c *gin.Context) {
	ip := a.getIP()
	t := strconv.Itoa(time.Now().Nanosecond())
	method := "WriteLogsToFile"
	a.fileLogger.WithFields(logrus.Fields{"log_time": t, "log_method": method, "log_ip": ip}).Error("写入到文件的错误日志")
	c.JSON(http.StatusOK, gin.H{
		"message": "OK",
	})
}
func (a *app) WriteLogsToCls(c *gin.Context) {
	ip := a.getIP()
	t := strconv.Itoa(time.Now().Nanosecond())
	method := "WriteLogsToCls"
	a.clsLogger.WithFields(logrus.Fields{"log_time": t, "log_method": method, "log_ip": ip}).Error("写入到日志中心的错误日志")
	c.JSON(http.StatusOK, gin.H{
		"message": "OK",
	})
}
func (a *app) Ping(c *gin.Context) {
	ip := a.getIP()
	c.JSON(http.StatusOK, gin.H{
		"ip": ip,
	})
}
func (a *app) Health(c *gin.Context) {
	c.JSON(http.StatusOK, gin.H{
		"message": "OK",
	})
}

func (a *app) getIP() string {
	var ip string
	addrs, err := net.InterfaceAddrs()
	if err != nil {
		a.fileLogger.Error(err)
		return ""
	}
	for _, v := range addrs {
		if ipnet, ok := v.(*net.IPNet); ok && !ipnet.IP.IsLoopback() {
			if ipnet.IP.To4() != nil {
				ip = ipnet.IP.String()
				break
			}
		}
	}
	return ip
}
```

对业务逻辑接口进行包装，统计接口点击的数据middleware.go：

```go
package app
import (
	"context"
	"github.com/gin-gonic/gin"
	"go.opentelemetry.io/otel/metric"
)
type middleApp struct {
	writeToFileCounter metric.Int64Counter
	writeToClsCounter  metric.Int64Counter
	next               AppInterface
}
func NewMiddleApp(writeToFileCounter metric.Int64Counter, writeToClsCounter metric.Int64Counter, next AppInterface) AppInterface {
	return &middleApp{
		writeToFileCounter: writeToFileCounter,
		writeToClsCounter:  writeToClsCounter,
		next:               next,
	}
}
func (a *middleApp) WriteLogsToFile(c *gin.Context) {
	a.writeToFileCounter.Add(context.Background(), 1)
	a.next.WriteLogsToFile(c)
}
func (a *middleApp) WriteLogsToCls(c *gin.Context) {
	a.writeToClsCounter.Add(context.Background(), 1)
	a.next.WriteLogsToCls(c)
}
func (a *middleApp) Ping(c *gin.Context) {
	a.next.Ping(c)
}
func (a *middleApp) Health(c *gin.Context) {
	a.next.Health(c)
}
```

入口main函数，进行资源初始化，监控端口提供接口服务main.go：

```go
package main
import (
	"ClusterManagement/app"
	"ClusterManagement/config"
	"ClusterManagement/logs"
	"context"
	"fmt"
	"github.com/gin-gonic/gin"
	"github.com/prometheus/client_golang/prometheus/promhttp"
	"github.com/sirupsen/logrus"
	"go.opentelemetry.io/otel"
	"go.opentelemetry.io/otel/attribute"
	"go.opentelemetry.io/otel/exporters/prometheus"
	metricapi "go.opentelemetry.io/otel/metric"
	"go.opentelemetry.io/otel/sdk/metric"
	"go.opentelemetry.io/otel/sdk/resource"
	semconv "go.opentelemetry.io/otel/semconv/v1.20.0"
	"io"
	"log"
	"net/http"
)
func initRouter(r *gin.Engine, app app.AppInterface) {
	r.GET("write/file", app.WriteLogsToFile)
	r.GET("write/cls", app.WriteLogsToCls)
	r.GET("/health", app.Health)
	r.GET("/ping", app.Ping)
}
func initProvider() (func(ctx context.Context) error, error) {
	ctx := context.Background()
	//初始化资源信息和属性
	res, err := resource.New(
		ctx, resource.WithOS(), resource.WithHost(),
		resource.WithAttributes(
			semconv.ServiceName("management"),
			semconv.ServiceVersion("1.0.0"),
			attribute.String("env", "dev"),
			attribute.String("author", "yangshuangxin"),
		),
	)
	if err != nil {
		log.Fatal(err)
	}
	//初始化Exporter与Reader
	var metricReader metric.Reader
	{
		metricReader, err = prometheus.New()
		if err != nil {
			log.Fatal(err)
		}
	}
	// 初始化Provider
	provider := metric.NewMeterProvider(metric.WithResource(res), metric.WithReader(metricReader))
	// 设置全局provider
	otel.SetMeterProvider(provider)
	return provider.Shutdown, nil
}

func initLogger(writer io.Writer) *logrus.Entry {
	logger := logrus.New()
	logger.SetFormatter(&logrus.JSONFormatter{})
	logger.SetOutput(writer)
	return logrus.NewEntry(logger)
}
func initCounter(meterName string, counterName string, desc string) metricapi.Int64Counter {
	meter := otel.Meter(meterName)
	counter, err := meter.Int64Counter(counterName, metricapi.WithDescription(desc))
	if err != nil {
		log.Fatal(err)
	}
	return counter
}

// 启动prometheus 指标输出端点
func serveMetrics() {
	http.Handle("/metrics", promhttp.Handler())
	err := http.ListenAndServe(":2223", nil)
	if err != nil {
		log.Fatal(err)
	}
}
func main() {
	shutdown, err := initProvider()
	if err != nil {
		log.Fatal(err)
	}
	defer shutdown(context.Background())
	writeToFileCounter := initCounter("management-metrics", "WriteToFile", "写入文件次数")
	writeToClsCounter := initCounter("management-metrics", "WriteToCls", "写入cls次数")

	conf := config.GetConf()
	fileWriter := logs.NewFileWriter(conf)
	defer fileWriter.Flush()
	clsWriter := logs.NewClsWriter(conf)
	defer clsWriter.Flush()

	app1 := app.NewApp(conf, initLogger(fileWriter), initLogger(clsWriter))
	middleApp := app.NewMiddleApp(writeToFileCounter, writeToClsCounter, app1)
	r := gin.Default()
	initRouter(r, middleApp)
	go serveMetrics()
	// 8080
	r.Run()

	fmt.Printf("%+v", config.GetConf())
}
```
